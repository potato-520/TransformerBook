4. Python / NumPy 完整 Transformer Block 实现

import numpy as np

class TransformerDecoderBlock:
    def __init__(self, d_model=64, n_heads=4, hidden_dim=172):
        self.d_model = d_model
        
        self.norm1_gamma = np.ones(d_model)
        self.norm2_gamma = np.ones(d_model)
        
        self.w_q = np.random.randn(d_model, d_model) * 0.02
        self.w_k = np.random.randn(d_model, d_model) * 0.02
        self.w_v = np.random.randn(d_model, d_model) * 0.02
        self.w_o = np.random.randn(d_model, d_model) * 0.02
        
        self.w_gate = np.random.randn(d_model, hidden_dim) * 0.02
        self.w_up   = np.random.randn(d_model, hidden_dim) * 0.02
        self.w_down = np.random.randn(hidden_dim, d_model) * 0.02

    def _rmsnorm(self, x, gamma):
        rms = np.sqrt(np.mean(x ** 2, axis=-1, keepdims=True) + 1e-6)
        return (x / rms) * gamma

    def forward(self, x):
        # 1. Attention 子层
        norm_x1 = self._rmsnorm(x, self.norm1_gamma)
        
        q = np.dot(norm_x1, self.w_q)
        k = np.dot(norm_x1, self.w_k)
        v = np.dot(norm_x1, self.w_v)
        
        scores = np.matmul(q, k.swapaxes(-1, -2)) / np.sqrt(self.d_model)
        mask = np.triu(np.ones_like(scores, dtype=bool), k=1)
        scores = np.where(mask, -1e9, scores)
        attn = np.exp(scores - np.max(scores, axis=-1, keepdims=True))
        attn = attn / np.sum(attn, axis=-1, keepdims=True)
        
        attn_out = np.dot(np.matmul(attn, v), self.w_o)
        x = x + attn_out  # 残差 1
        
        # 2. FFN 子层
        norm_x2 = self._rmsnorm(x, self.norm2_gamma)
        gate = np.dot(norm_x2, self.w_gate)
        silu_gate = gate / (1.0 + np.exp(-gate))
        up = np.dot(norm_x2, self.w_up)
        ffn_out = np.dot(silu_gate * up, self.w_down)
        
        x = x + ffn_out   # 残差 2
        return x

block = TransformerDecoderBlock(d_model=64, n_heads=4, hidden_dim=172)
fake_in = np.random.randn(1, 6, 64)
out = block.forward(fake_in)
print("输入 Shape:", fake_in.shape)
print("通过完整 Decoder Block 后 Shape:", out.shape)

5. 打开黑盒：真实工程细节 (堆叠 32 层时的吞吐与显存)

大模型流水线堆叠：
在 LLaMA-3 8B 中，上述完全相同的 Decoder Block 在垂直方向连续堆叠了整整 32 层！
每一层的输出张量直接作为下一层的输入张量，在 32 层之间以极高的吞吐量持续流转提炼，直到最后送入 Final RMSNorm 和 LM Head 输出 Logits。
